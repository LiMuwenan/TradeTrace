> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 打通「选图 → 离线 OCR → 同花顺解析 → 逐笔确认 → 原子入库」闭环：未确认不入库、重复导入不重复入库、飞行模式可用。

**Architecture:** 三层：`src/data/import/*`（纯函数：图片归一、草稿构建、入库计划）→ `src/data/state.ts`（唯一的批量入库纯函数，负责建股票与 `dedupeKey` 幂等）→ `src/app/import/*`（选图页 / 确认页 / 结果页，只做展示与交互）。识别草稿只存内存（`ImportProvider`），不落盘；只有用户点「确认入库」才调用 `actions.importTransactions`。

**Tech Stack:** TypeScript strict / React Native 0.88 / Expo SDK 58（expo-router、expo-image-picker、expo-image-manipulator、expo-crypto）/ Vitest（node 环境，只测纯函数）

**关键约束：**
- 页面不得直接 `fetch`，也不得直接调 `storage.ts`；数据写入只经 `TradeProvider.actions`。

- 金额/股数一律使用领域口径：`shares` 正=买入、负=卖出，股息 `shares=0`、`dividend` 可为负。

```ts
/** OCR 导入：已有股票直接引用，否则在目标账户下按名称/代码新建 */
export interface ImportStockSpec {
  accountId: string
  name: string
  code: string
}

export interface ImportItem {
  stock: { id: string } | { create: ImportStockSpec }
  /** 不含 stockId 的交易字段；source 缺省为 ocr */
  input: Omit<TransactionInput, 'stockId'>
}

export interface ImportOutcome {
  state: TradeState
  inserted: number
  duplicated: number
  createdStocks: number
}

/**
 * 批量入库（OCR 导入的唯一写入口）。
 * 1. 同一账户下同名同代码的股票复用，不重复创建；
 * 2. 以 dedupeKey 幂等：库内已有或批内重复的条目直接跳过；
 * 3. 没有任何写入时原样返回，避免空导入推进 revision。
 */
export function importTransactions(
  state: TradeState,
  items: ImportItem[],
  now: string,
): ImportOutcome {
  const stocks = [...state.stocks]
  const keys = new Set(
    state.transactions.filter((item) => !item.deletedAt).map((item) => item.dedupeKey),
  )
  const appended: TransactionTrade[] = []
  let inserted = 0
  let duplicated = 0
  let createdStocks = 0

  const resolveStock = (item: ImportItem): Stock | null => {
    if ('id' in item.stock) {
      return stocks.find((stock) => stock.id === item.stock.id && !stock.deletedAt) ?? null
    }

    const spec = item.stock.create
    const name = spec.name.trim()
    const code = spec.code.trim()
    if (!name) return null

    const existing = stocks.find(
      (stock) =>
        !stock.deletedAt &&
        stock.accountId === spec.accountId &&
        stock.name === name &&
        stock.code === code,
    )
    if (existing) return existing

    return {
      id: createId(),
      accountId: spec.accountId,
      name,
      code,
      createdAt: now,
      updatedAt: now,
    }
  }

  for (const item of items) {
    const stock = resolveStock(item)
    if (!stock) continue

    const dedupeKey = computeDedupeKey(stock.code, item.input)
    if (keys.has(dedupeKey)) {
      duplicated += 1
      continue
    }
    keys.add(dedupeKey)

    const isNewStock = !stocks.some((candidate) => candidate.id === stock.id)
    if (isNewStock) {
      stocks.push(stock)
      createdStocks += 1
    }

    appended.push({
      ...item.input,
      stockId: stock.id,
      id: createId(),
      createdAt: now,
      updatedAt: now,
      source: item.input.source ?? 'ocr',
      dedupeKey,
    })
    inserted += 1
  }

  if (appended.length === 0) {
    return { state, inserted: 0, duplicated, createdStocks: 0 }
  }

  return {
    state: touch(state, now, {
      stocks,
      transactions: [...state.transactions, ...appended],
    }),
    inserted,
    duplicated,
    createdStocks,
  }
}
```

```ts
import * as Crypto from 'expo-crypto'
import { manipulate, SaveFormat } from 'expo-image-manipulator'

/** 长边上限：与 OCR POC 的预处理保持一致，避免不同设备上识别结果漂移 */
const MAX_EDGE = 1600

export interface PreparedImage {
  uri: string
  width: number
  height: number
  /** 归一下采样后的内容哈希，写入 sourceRef 用于溯源 */
  imageHash: string
}

/**
 * 归一下采样并计算内容哈希。
 * 同一张截图重复选择会得到相同 imageHash；交易是否重复由 dedupeKey 判定，二者互不替代。
 */
export async function prepareImage(uri: string): Promise<PreparedImage> {
  const rendered = await manipulate(uri, {
    maxWidth: MAX_EDGE,
    maxHeight: MAX_EDGE,
  }).renderAsync()
  const saved = await rendered.saveAsync({
    format: SaveFormat.JPEG,
    compress: 0.85,
    base64: true,
  })
  const imageHash = await Crypto.digestStringAsync(
    Crypto.CryptoDigestAlgorithm.SHA256,
    saved.base64 ?? saved.uri,
  )
  return { uri: saved.uri, width: saved.width, height: saved.height, imageHash }
}
```

说明（SDK 58 已核对官方文档）：
- `manipulate(source, options)` 的第二参数是 `ImageLoadOptions`（`maxWidth` / `maxHeight` 在解码阶段下采样，省内存）。
- `saveAsync` 的 `SaveOptions` 支持 `base64: true`，`ImageResult` 会返回 `base64`、`width`、`height`、`uri`。
- 不能用已废弃的 `manipulateAsync`。

```ts
import type { MarketStock } from '@/data/api/market'

import { createId } from '../ids'
import type { Stock, TransactionTrade, TransactionInput } from '../types'
import { computeDedupeKey } from '../types'
import { candidateToInput, normalizeStockName } from './normalize'
import type { ParsedRow } from './types'

/** 一张截图的解析结果 */
export interface ParsedImage {
  imageHash: string
  templateId: string
  rows: ParsedRow[]
}

/** 确认页上的一行草稿（只存在内存，不落盘） */
export interface DraftRow {
  draftId: string
  row: ParsedRow
  imageHash: string
  templateId: string
  /** 归一化后的名称，用于展示与新建股票 */
  name: string
  /** 命中本机股票时的 ID */
  stockId: string | null
  /** 字典建议代码，或用户手工填写；本机没有该股票时用于新建 */
  code: string
  /** 本机与字典都没命中，需要人工选择股票或填写代码 */
  needsManualCode: boolean
  /** 与库内或本批已存在的 dedupeKey 相同 */
  duplicate: boolean
  selected: boolean
  warnings: string[]
}

export interface DraftContext {
  stocks: Stock[]
  dictionary: MarketStock[]
  transactions: TransactionTrade[]
}

/** 确认页展示的最终统计 */
export interface ImportSummary {
  inserted: number
  duplicates: number
  invalid: number
  createdStocks: number
}

export interface PlannedItem {
  stock: { id: string } | { create: { accountId: string; name: string; code: string } }
  input: Omit<TransactionInput, 'stockId'>
}

export interface ImportPlan {
  items: PlannedItem[]
  duplicates: number
  invalid: number
  /** 需要新建的股票数（同名单只计一次） */
  newStocks: number
}

function matchLocal(name: string, stocks: Stock[]): Stock | null {
  const exact = stocks.find((stock) => !stock.deletedAt && stock.name === name)
  if (exact) return exact
  return (
    stocks.find((stock) => !stock.deletedAt && normalizeStockName(stock.name) === name) ?? null
  )
}

function matchDictionary(name: string, dictionary: MarketStock[]): MarketStock | null {
  const exact = dictionary.find((item) => item.name === name)
  if (exact) return exact
  return dictionary.find((item) => normalizeStockName(item.name) === name) ?? null
}

/** 解析结果 → 确认页草稿：补股票匹配、字典建议与重复判定，不修改任何数据 */
export function buildDrafts(images: ParsedImage[], context: DraftContext): DraftRow[] {
  const keys = new Set(
    context.transactions.filter((item) => !item.deletedAt).map((item) => item.dedupeKey),
  )
  const drafts: DraftRow[] = []

  for (const image of images) {
    for (const row of image.rows) {
      const name = normalizeStockName(row.name)
      const local = matchLocal(name, context.stocks)
      const suggestion = matchDictionary(name, context.dictionary)
      const code = local?.code ?? suggestion?.code ?? ''
      const dedupeKey = computeDedupeKey(code, row)
      const duplicate = keys.has(dedupeKey)
      if (!duplicate) keys.add(dedupeKey)

      const warnings = [...row.warnings]
      if (!local && !suggestion) {
        warnings.push('未在本地股票或行情字典中找到该名称，请手工选择股票或填写 6 位代码')
      }

      drafts.push({
        draftId: createId(),
        row,
        imageHash: image.imageHash,
        templateId: image.templateId,
        name,
        stockId: local?.id ?? null,
        code,
        needsManualCode: !local && !suggestion,
        duplicate,
        selected: !duplicate,
        warnings,
      })
    }
  }

  return drafts
}

/**
 * 草稿 → 入库计划：只统计、不写数据。
 * 重复行无条件排除（即使被勾选），缺少股票与代码的计入 invalid。
 */
export function planImport(
  drafts: DraftRow[],
  options: { accountId: string; stocks: Stock[] },
): ImportPlan {
  const items: PlannedItem[] = []
  const newStockKeys = new Set<string>()
  let duplicates = 0
  let invalid = 0

  for (const draft of drafts) {
    if (draft.duplicate) {
      duplicates += 1
      continue
    }
    if (!draft.selected) continue

    const code = draft.code.trim()
    if (!draft.stockId) {
      if (!/^\d{6}$/.test(code)) {
        invalid += 1
        continue
      }
      const exists = options.stocks.some(
        (stock) =>
          !stock.deletedAt &&
          stock.accountId === options.accountId &&
          stock.name === draft.name &&
          stock.code === code,
      )
      if (!exists) newStockKeys.add(`${options.accountId}|${draft.name}|${code}`)
    }

    const { stockId, ...input } = candidateToInput(draft.row, draft.stockId ?? '', {
      imageHash: draft.imageHash,
      templateId: draft.templateId,
      lineNo: draft.row.rowIndex,
    })
    void stockId

    items.push({
      stock: draft.stockId
        ? { id: draft.stockId }
        : { create: { accountId: options.accountId, name: draft.name, code } },
      input,
    })
  }

  return { items, duplicates, invalid, newStocks: newStockKeys.size }
}
```

```tsx
import { createContext, useContext, useMemo, useState } from 'react'
import type { ReactNode } from 'react'

import type { DraftRow, ImportSummary } from './import/draft'

interface ImportContextValue {
  drafts: DraftRow[]
  setDrafts: (rows: DraftRow[]) => void
  summary: ImportSummary | null
  setSummary: (summary: ImportSummary | null) => void
  reset: () => void
}

const ImportContext = createContext<ImportContextValue | null>(null)

/** 导入草稿只放内存：未确认的识别结果不得写入本地数据 */
export function ImportProvider({ children }: { children: ReactNode }) {
  const [drafts, setDrafts] = useState<DraftRow[]>([])
  const [summary, setSummary] = useState<ImportSummary | null>(null)

  const value = useMemo<ImportContextValue>(
    () => ({
      drafts,
      setDrafts,
      summary,
      setSummary,
      reset: () => {
        setDrafts([])
        setSummary(null)
      },
    }),
    [drafts, summary],
  )

  return <ImportContext.Provider value={value}>{children}</ImportContext.Provider>
}

export function useImportDraft(): ImportContextValue {
  const value = useContext(ImportContext)
  if (!value) throw new Error('useImportDraft 必须在 ImportProvider 内使用')
  return value
}
```

```tsx
                <RowButton
                  key={stock.id}
                  onPress={() => {
                    chooseStock(stockSheetDraft.draftId, stock)
                    setStockSheetDraft(null)
                  }}
                >
                  <View style={styles.accountRow}>
                    <Text style={styles.accountName}>{stock.name}</Text>
                    <Text style={styles.stockCode}>{stock.code || '无代码'}</Text>
                  </View>
                </RowButton>
              ))}
            <Field
              label="新建股票代码（6 位）"
              value={codeInput}
              onChangeText={setCodeInput}
              keyboardType="numeric"
              maxLength={6}
            />
            {sheetError ? <Text style={styles.error}>{sheetError}</Text> : null}
            <Button
              label="按识别名称新建"
              variant="secondary"
              onPress={() => {
                const code = codeInput.trim()
                if (!/^\d{6}$/.test(code)) {
                  setSheetError('请输入 6 位股票代码')
                  return
                }
                chooseNewStock(stockSheetDraft.draftId, code)
                setStockSheetDraft(null)
              }}
            />
          </View>
        ) : null}
      </Sheet>
    </Screen>
  )
}
```

## 验收对照（来自任务表）

| 完成标准 | 落地 |
|---|---|
| 飞行模式可用 | OCR 在本机、字典读缓存、无任何网络请求；Task 6 与 Task 9 Step 3 |
| 重复导入不重复入库 | `dedupeKey` 双层保障：Task 3 草稿判重（默认不勾选）+ Task 1 入库判重（纯函数用例） |
| 未确认不入库 | 草稿只存 `ImportProvider` 内存；只有 Task 7 的「确认入库」调用 `actions.importTransactions` |
| 依赖 1–6 | Task 1 依赖 v2 模型；Task 3/6 依赖解析器；Task 4 依赖引擎与行情服务；本计划不重复实现它们 |

## 风险与注意事项

- **`mlkit-engine` 的 Web 打包风险**：Task 6 Step 2 已给出处理顺序（先记录导出失败原文，再按平台拆分）。不得为了让 Web 构建通过而删除 Android 识别分支。
- **草稿不持久化**：应用被杀后草稿丢失，这是「未确认不入库」的必然代价，不在本期做草稿恢复。
- **`dedupeKey` 碰撞**：同股票同日期同方向同价格同数量同手续费的两笔会被视为同一笔（v2 既有口径），确认页会把第二笔标为重复，需人工补录。
- **账户选择**：已有股票不受所选账户影响；只有新建股票归属所选账户，结果页需向用户说明这一点。
- **不要新增跨项目请求**：股票字典只经任务 4 的 `loadStockDictionary`（内部走 `/api/v1/market/stocks/snapshot`），页面不得直接 `fetch`。
