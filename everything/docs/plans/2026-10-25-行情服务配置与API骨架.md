```ts
import AsyncStorage from '@react-native-async-storage/async-storage'

/** 行情服务配置：地址由用户在设置页填写，不在代码里写死 */
export interface MarketConfig {
  /** 形如 http://192.168.1.10:8000，不含末尾斜杠与 /api/v1 */
  baseUrl: string
  timeoutMs: number
  cacheEnabled: boolean
  /** 最近一次连接测试成功时间（ISO） */
  lastConnectedAt: string
  /** 最近一次连接测试结果 */
  lastHealthStatus: 'ok' | 'fail' | ''
}

const STORAGE_KEY = 'tradetrace/market/v1'
export const DEFAULT_TIMEOUT_MS = 8000

export const DEFAULT_MARKET_CONFIG: MarketConfig = {
  baseUrl: '',
  timeoutMs: DEFAULT_TIMEOUT_MS,
  cacheEnabled: true,
  lastConnectedAt: '',
  lastHealthStatus: '',
}

export function normalizeMarketUrl(raw: string): string {
  const trimmed = raw.trim().replace(/\/+$/, '')
  return trimmed.replace(/\/api\/v1$/i, '')
}

/** 校验并返回归一化地址；Android 的 localhost 指向手机自身，直接拦下 */
export function validateMarketUrl(raw: string): string {
  const url = normalizeMarketUrl(raw)
  if (!url) throw new Error('请填写行情服务地址')
  if (!/^https?:\/\//i.test(url)) throw new Error('地址必须以 http:// 或 https:// 开头')
  if (/^https?:\/\/localhost/i.test(url)) {
    throw new Error('Android 端 localhost 指向手机自身，请填写电脑的局域网 IP')
  }
  return url
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value)
}

export async function loadMarketConfig(): Promise<MarketConfig> {
  try {
    const raw = await AsyncStorage.getItem(STORAGE_KEY)
    if (!raw) return { ...DEFAULT_MARKET_CONFIG }
    const parsed: unknown = JSON.parse(raw)
    if (!isRecord(parsed)) return { ...DEFAULT_MARKET_CONFIG }
    return {
      baseUrl: typeof parsed.baseUrl === 'string' ? parsed.baseUrl : '',
      timeoutMs: typeof parsed.timeoutMs === 'number' ? parsed.timeoutMs : DEFAULT_TIMEOUT_MS,
      cacheEnabled: parsed.cacheEnabled !== false,
      lastConnectedAt: typeof parsed.lastConnectedAt === 'string' ? parsed.lastConnectedAt : '',
      lastHealthStatus:
        parsed.lastHealthStatus === 'ok' || parsed.lastHealthStatus === 'fail'
          ? parsed.lastHealthStatus
          : '',
    }
  } catch (error) {
    console.warn('[TradeTrace] 读取行情服务配置失败', error)
    return { ...DEFAULT_MARKET_CONFIG }
  }
}

export async function saveMarketConfig(next: MarketConfig): Promise<void> {
  await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(next))
}
```

```ts
/** 行情请求错误分类：页面据此给出可执行提示，不暴露底层细节 */
export type MarketErrorKind = 'config' | 'network' | 'timeout' | 'http' | 'business' | 'parse'

export class MarketApiError extends Error {
  readonly kind: MarketErrorKind
  readonly code: string
  readonly status: number

  constructor(kind: MarketErrorKind, message: string, options: { code?: string; status?: number } = {}) {
    super(message)
    this.name = 'MarketApiError'
    this.kind = kind
    this.code = options.code ?? ''
    this.status = options.status ?? 0
  }
}

export interface RequestOptions {
  timeoutMs?: number
  signal?: AbortSignal
}

const DEFAULT_TIMEOUT_MS = 8000

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value)
}

/**
 * 统一 JSON 请求：超时、网络、HTTP、业务、解析五类错误都归一为 MarketApiError。
 * baseUrl 由设置页提供，path 必须是 /api/v1 开头的绝对路径。
 */
export async function requestJson<T>(
  baseUrl: string,
  path: string,
  options: RequestOptions = {},
): Promise<T> {
  const base = baseUrl.trim().replace(/\/+$/, '')
  if (!base) {
    throw new MarketApiError('config', '尚未配置行情服务地址，请到设置页填写')
  }
  const timeoutMs = options.timeoutMs ?? DEFAULT_TIMEOUT_MS
  const controller = new AbortController()
  const timer = setTimeout(() => controller.abort(), timeoutMs)

  try {
    const response = await fetch(`${base}${path}`, {
      method: 'GET',
      headers: { Accept: 'application/json' },
      signal: options.signal ?? controller.signal,
    })
    const text = await response.text()
    let body: unknown = null
    if (text) {
      try {
        body = JSON.parse(text)
      } catch {
        if (!response.ok) {
          throw new MarketApiError('http', `服务返回 ${response.status}`, { status: response.status })
        }
        throw new MarketApiError('parse', '服务返回内容不是合法 JSON')
      }
    }
    if (!response.ok) {
      const payload = isRecord(body) ? body : null
      const code = payload && typeof payload.code === 'string' ? payload.code : `HTTP_${response.status}`
      const message =
        payload && typeof payload.message === 'string'
          ? payload.message
          : `请求失败（HTTP ${response.status}）`
      throw new MarketApiError('business', message, { code, status: response.status })
    }
    return body as T
  } catch (error) {
    if (error instanceof MarketApiError) throw error
    if (error instanceof Error && error.name === 'AbortError') {
      throw new MarketApiError('timeout', '连接超时：请确认服务已启动且手机与电脑在同一局域网')
    }
    throw new MarketApiError('network', '无法连接行情服务：请检查地址、Wi-Fi 与防火墙')
  } finally {
    clearTimeout(timer)
  }
}
```

```ts
    .filter(([, value]) => value !== undefined && value !== '')
    .map(([key, value]) => `${key}=${encodeURIComponent(String(value))}`)
  return pairs.length ? `?${pairs.join('&')}` : ''
}

  const path = `/api/v1/market/stocks${query({ q, limit: options.limit })}`
```

```ts
import AsyncStorage from '@react-native-async-storage/async-storage'

const PREFIX = 'tradetrace/market/cache/v1/'
export const SNAPSHOT_TTL_MS = 7 * 24 * 60 * 60 * 1000
export const KLINE_TTL_MS = 12 * 60 * 60 * 1000

export interface CacheEntry<T> {
  savedAt: string
  data: T
}

function storageKey(key: string): string {
  return `${PREFIX}${key}`
}

export async function readCache<T>(key: string): Promise<CacheEntry<T> | null> {
  try {
    const raw = await AsyncStorage.getItem(storageKey(key))
    if (!raw) return null
    const parsed: unknown = JSON.parse(raw)
    if (typeof parsed !== 'object' || parsed === null) return null
    const entry = parsed as { savedAt?: unknown; data?: unknown }
    if (typeof entry.savedAt !== 'string' || entry.data === undefined) return null
    return { savedAt: entry.savedAt, data: entry.data as T }
  } catch (error) {
    console.warn('[TradeTrace] 读取行情缓存失败', error)
    return null
  }
}

export async function writeCache<T>(key: string, data: T, now: string): Promise<void> {
  try {
    await AsyncStorage.setItem(storageKey(key), JSON.stringify({ savedAt: now, data }))
  } catch (error) {
    console.warn('[TradeTrace] 写入行情缓存失败', error)
  }
}

export function isFresh(savedAt: string, ttlMs: number, nowMs: number = Date.now()): boolean {
  const saved = Date.parse(savedAt)
  if (!Number.isFinite(saved)) return false
  return nowMs - saved <= ttlMs
}

export function snapshotKey(): string {
  return 'stocks/snapshot'
}

export function klineKey(code: string, from: string, to: string): string {
  return `kline/${code}/${from || 'all'}_${to || 'all'}`
}
```

```ts
import type { MarketApiError } from './api/client'
import { fetchHealth, fetchKline, fetchSnapshot, searchStocks } from './api/market'
import type { KlineResponse, MarketStock } from './api/market'
import {
  KLINE_TTL_MS,
  SNAPSHOT_TTL_MS,
  isFresh,
  klineKey,
  readCache,
  snapshotKey,
  writeCache,
} from './market-cache'
import { loadMarketConfig } from './market-config'

export interface StockDictionaryResult {
  stocks: MarketStock[]
  generatedAt: string
  fromCache: boolean
  /** true 表示本次展示的是过期缓存，页面必须提示「数据可能不是最新」 */
  stale: boolean
  /** 降级原因；网络正常时为空 */
  error: string
}

export interface KlineResult {
  data: KlineResponse
  fromCache: boolean
  stale: boolean
  error: string
}

function errorText(error: unknown): string {
  if (error instanceof Error) return error.message
  return '未知错误'
}

async function config() {
  return loadMarketConfig()
}

/**
 * 股票字典：TTL 内直接读缓存；否则联网刷新并回写。
 * 联网失败时回退旧缓存（stale=true），无缓存则抛出错误。
 */
export async function loadStockDictionary(options: { force?: boolean } = {}): Promise<StockDictionaryResult> {
  const cfg = await config()
  const cached = await readCache<{ generatedAt: string; stocks: MarketStock[] }>(snapshotKey())
  const fresh = cached !== null && isFresh(cached.savedAt, SNAPSHOT_TTL_MS)
  if (cached && fresh && !options.force) {
    return {
      stocks: cached.data.stocks,
      generatedAt: cached.data.generatedAt,
      fromCache: true,
      stale: false,
      error: '',
    }
  }
  try {
    const snapshot = await fetchSnapshot(cfg.baseUrl, { timeoutMs: cfg.timeoutMs })
    if (cfg.cacheEnabled) {
      await writeCache(snapshotKey(), { generatedAt: snapshot.generatedAt, stocks: snapshot.stocks }, new Date().toISOString())
    }
    return {
      stocks: snapshot.stocks,
      generatedAt: snapshot.generatedAt,
      fromCache: false,
      stale: false,
      error: '',
    }
  } catch (error) {
    if (cached) {
      return {
        stocks: cached.data.stocks,
        generatedAt: cached.data.generatedAt,
        fromCache: true,
        stale: true,
        error: errorText(error),
      }
    }
    throw error
  }
}

export async function searchMarketStocks(keyword: string): Promise<MarketStock[]> {
  const cfg = await config()
  return searchStocks(cfg.baseUrl, keyword, { timeoutMs: cfg.timeoutMs })
}

/** 日K：TTL 内读缓存；失败回退缓存并标记 stale */
export async function loadKline(
  code: string,
  from: string,
  to: string,
): Promise<KlineResult> {
  const cfg = await config()
  const key = klineKey(code, from, to)
  const cached = await readCache<KlineResponse>(key)
  if (cached && isFresh(cached.savedAt, KLINE_TTL_MS)) {
    return { data: cached.data, fromCache: true, stale: false, error: '' }
  }
  try {
    const data = await fetchKline(cfg.baseUrl, code, { from, to, timeoutMs: cfg.timeoutMs })
    if (cfg.cacheEnabled) await writeCache(key, data, new Date().toISOString())
    return { data, fromCache: false, stale: false, error: '' }
  } catch (error) {
    if (cached) {
      return { data: cached.data, fromCache: true, stale: true, error: errorText(error) }
    }
    throw error
  }
}

/** 连接测试：成功后回写 lastConnectedAt/lastHealthStatus */
export async function testMarketConnection(baseUrl?: string): Promise<{ ok: boolean; message: string }> {
  const cfg = await loadMarketConfig()
  const target = (baseUrl ?? cfg.baseUrl).trim()
  try {
    const health = await fetchHealth(target, { timeoutMs: cfg.timeoutMs })
    if (health.schemaVersion !== 1 || health.status !== 'ok') {
      return { ok: false, message: '服务响应异常：请确认这是 TradeTrace 行情服务' }
    }
    const { saveMarketConfig } = await import('./market-config')
    await saveMarketConfig({
      ...cfg,
      baseUrl: target.replace(/\/+$/, ''),
      lastConnectedAt: new Date().toISOString(),
      lastHealthStatus: 'ok',
    })
    return { ok: true, message: `连接成功（服务版本 ${health.version}）` }
  } catch (error) {
    const err = error as MarketApiError | Error
    return { ok: false, message: errorText(err) }
  }
}
```

> 说明：错误对象由 `requestJson` 抛出，`message` 已是中文可展示文案，页面直接显示即可。

```ts
  async function persistMarket(): Promise<MarketConfig | null> {
    if (!market) return null
    try {
      const baseUrl = validateMarketUrl(marketUrl)
      const next: MarketConfig = { ...market, baseUrl }
      await saveMarketConfig(next)
      setMarket(next)
      setMarketUrl(next.baseUrl)
      return next

      showNotice(error instanceof Error ? error.message : '地址不合法', false)
      return null
    }

  async function testMarket() {
    const saved = await persistMarket()
    if (!saved) return
    setMarketBusy('test')
    const result = await testMarketConnection(saved.baseUrl)
    setMarketBusy(null)
    if (result.ok) {
      const reloaded = await loadMarketConfig()
      setMarket(reloaded)
    }
    showNotice(result.message, result.ok)
  }

  async function refreshDictionary() {
    const saved = await persistMarket()
    if (!saved) return
    setMarketBusy('refresh')
    try {

      setDictInfo('')
```
