```python
# -*- coding: utf-8 -*-
"""统一预处理：长边限制 1600；深色截图反转为浅底深字。

产物：assets/ocr/_poc/images/<原名>.jpg（供所有引擎共用同一输入）
"""
import json
import sys
from pathlib import Path

from PIL import Image, ImageStat

SRC = Path("assets/ocr/同花顺")
OUT = Path("assets/ocr/_poc/images")
MAX_SIDE = 1600
DARK_THRESHOLD = 110.0  # 平均亮度低于该值视为深色主题


def load_color(path: Path) -> Image.Image:
    img = Image.open(path).convert("RGB")
    w, h = img.size
    scale = min(1.0, MAX_SIDE / max(w, h))
    if scale < 1.0:
        img = img.resize((int(w * scale), int(h * scale)))
    return img


def normalize_theme(img: Image.Image) -> tuple[Image.Image, bool]:
    gray = img.convert("L")
    mean = ImageStat.Stat(gray).mean[0]
    if mean < DARK_THRESHOLD:
        return Image.eval(img, lambda v: 255 - v), True
    return img, False


def main() -> None:
    OUT.mkdir(parents=True, exist_ok=True)
    manifest = []
    for path in sorted(SRC.glob("*.jpg")):
        if path.stem.endswith("(3)"):      # 干扰样本
            continue
        img, inverted = normalize_theme(load_color(path))
        target = OUT / path.name
        img.save(target, quality=95)
        manifest.append({"file": path.name, "inverted": inverted, "size": img.size})
    (Path("assets/ocr/_poc") / "preprocess.json").write_text(
        json.dumps(manifest, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    print(f"processed {len(manifest)} images")


if __name__ == "__main__":
    sys.exit(main())
```

- [ ] **Step 4: 写 `run_paddle.py`（输出 `OcrLine[]` 与耗时）**

```python
# -*- coding: utf-8 -*-
"""PaddleOCR 出图：每张图输出 OcrLine[]（text/x/y/width/height/confidence）+ 耗时。"""
import json
import time
from pathlib import Path

from paddleocr import PaddleOCR

IMAGES = Path("assets/ocr/_poc/images")
OUT = Path("assets/ocr/_poc/lines/paddle")


def to_lines(result) -> list[dict]:
    lines = []
    for entry in result or []:
        box, (text, score) = entry[0], entry[1]
        xs = [point[0] for point in box]
        ys = [point[1] for point in box]
        lines.append({
            "text": text,
            "x": min(xs),
            "y": min(ys),
            "width": max(xs) - min(xs),
            "height": max(ys) - min(ys),
            "confidence": float(score),
        })
    return lines


def main() -> None:
    OUT.mkdir(parents=True, exist_ok=True)
    ocr = PaddleOCR(use_angle_cls=False, lang="ch", show_log=False)
    timings = []
    files = sorted(IMAGES.glob("*.jpg"))
    for index, path in enumerate(files):
        start = time.perf_counter()
        result = ocr.ocr(str(path), cls=False)
        elapsed = time.perf_counter() - start
        lines = to_lines(result[0] if result else [])
        (OUT / f"{path.stem}.json").write_text(
            json.dumps({"file": path.name, "elapsedMs": round(elapsed * 1000), "lines": lines},
                       ensure_ascii=False, indent=2),
            encoding="utf-8",
        )
        if index == 0:
            timings.append({"file": path.name, "elapsedMs": round(elapsed * 1000), "warmup": True})
        else:
            timings.append({"file": path.name, "elapsedMs": round(elapsed * 1000), "warmup": False})
        print(f"{path.name}: {len(lines)} lines, {elapsed * 1000:.0f} ms")
    (Path("assets/ocr/_poc") / "paddle-timings.json").write_text(
        json.dumps(timings, ensure_ascii=False, indent=2), encoding="utf-8"
    )


if __name__ == "__main__":
    main()
```

```ts
import TextRecognition from '@react-native-ml-kit/text-recognition'

import type { OcrEngine, OcrLine } from '@/data/import/types'

/** ML Kit 中文文本识别适配器：仅 Android 可用 */
export const mlkitEngine: OcrEngine = {
  async recognize(imageUri: string): Promise<OcrLine[]> {
    const result = await TextRecognition.recognize(imageUri)
    const lines: OcrLine[] = []
    for (const block of result.blocks ?? []) {
      for (const line of block.lines ?? []) {
        const frame = line.frame
        if (!frame) continue
        lines.push({
          text: line.text ?? '',
          x: frame.left,
          y: frame.top,
          width: frame.width,
          height: frame.height,
          confidence: 1,
        })
      }
    }
    return lines
  },
}
```

> ML Kit 不提供逐行置信度，统一记 `1`；评分报告中必须注明「ML Kit 置信度不可比」，字段准确率仍按最终字段比对。

- [ ] **Step 1: 写评分脚本（默认跳过，不影响 CI）**

```ts
import { readFileSync, readdirSync, writeFileSync } from 'node:fs'
import { join } from 'node:path'

import { describe, expect, it } from 'vitest'

import { parseThsStatement } from './parse-ths'
import type { OcrLine } from './types'

interface ExpectedRow {
  type: string
  name: string
  date: string
  time: string
  price: number
  shares: number
  commission: number
  dividend: number | null
}

interface ExpectedFile {
  file: string
  accountHint?: string
  skippedKinds?: string[]
  rows: ExpectedRow[]
}

const root = process.env.OCR_POC_DIR
const runner = root ? it : it.skip

function readJson<T>(path: string): T {
  return JSON.parse(readFileSync(path, 'utf-8')) as T
}

/** 同一张图、同一天、同一方向、同一名称视为同一行，再比对字段 */
function keyOf(row: { type: string; name: string; date: string }): string {
  return `${row.type}|${row.name}|${row.date}`
}

runner('OCR POC：字段级准确率', () => {
  const dir = root as string
  const expectedFiles = readdirSync(join(dir, 'expected')).filter((name) => name.endsWith('.json'))
  const fields = ['type', 'name', 'date', 'time', 'price', 'shares', 'commission', 'dividend'] as const
  const stats: Record<string, { total: number; ok: number }> = {}
  for (const field of fields) stats[field] = { total: 0, ok: 0 }

  const engines = ['paddle', 'mlkit'] as const
  const summary: Record<string, unknown> = {}

  for (const engine of engines) {
    let rowTotal = 0
    let rowComplete = 0
    let fileMissing = 0
    const cases: unknown[] = []
    for (const expectedName of expectedFiles) {
      const expected = readJson<ExpectedFile>(join(dir, 'expected', expectedName))
      const linesPath = join(dir, 'lines', engine, `${expected.file.replace(/\.jpg$/, '')}.json`)
      let lines: OcrLine[] = []
      try {
        lines = readJson<{ lines: OcrLine[] }>(linesPath).lines
      } catch {
        fileMissing += 1
        continue
      }
      const parsed = parseThsStatement(lines, { imageWidth: 1600, today: '2026-10-25' })
      const found = new Map(parsed.candidates.map((row) => [keyOf(row), row]))
      for (const row of expected.rows) {
        rowTotal += 1
        const actual = found.get(keyOf(row))
        let allOk = Boolean(actual)
        for (const field of fields) {
          stats[field].total += 1
          const expectedValue = row[field]
          const actualValue = actual ? (actual as unknown as Record<string, unknown>)[field] ?? null : null
          const ok =
            typeof expectedValue === 'number' && typeof actualValue === 'number'
              ? Math.abs(expectedValue - actualValue) <= (field === 'price' || field === 'commission' || field === 'dividend' ? 0.02 : 0)
              : expectedValue === actualValue
          if (ok) stats[field].ok += 1
          else allOk = false
        }
        if (allOk) rowComplete += 1
        if (!allOk) cases.push({ expected: row, actual: actual ?? null })
      }
    }
    summary[engine] = {
      filesMissing: fileMissing,
      rows: rowTotal,
      rowComplete,
      rowCompleteRate: rowTotal ? rowComplete / rowTotal : 0,
      fields: Object.fromEntries(
        fields.map((field) => [field, stats[field].total ? stats[field].ok / stats[field].total : 0]),
      ),
      mismatches: cases,
    }
  }

  const target = join(dir, 'report.json')
  writeFileSync(target, JSON.stringify(summary, null, 2), 'utf-8')
  console.log(JSON.stringify(summary, null, 2))
  expect(Object.keys(summary)).toHaveLength(2)
})
```
